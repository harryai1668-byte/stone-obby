const WebSocket = require("ws"); const wss = new WebSocket.Server({ port: 8080 }); const players = new Map(); const rooms = new Map(); let nextId = 1; const skinCost = { Noob: 0, Red: 50, Ninja: 100, Gold: 200 }; /* ========================= HELPERS ========================= */ function send(ws, data) { if (ws.readyState === WebSocket.OPEN) { ws.send( JSON.stringify(data) ); } } function broadcast(room, data) { for (const p of room.players) { send(p.ws, data); } } function createRoomCode() { let code; do { code = Math.random() .toString(36) .substring(2, 7) .toUpperCase(); } while (rooms.has(code)); return code; } function updateRoom(room) { const readyCount = room.players.filter( p => p.ready ).length; let text; if ( room.players.length < 2 ) { text = room.players.length+ "/2 • Đang chờ đối thủ..."; } else { text = "2/2 • READY "+ readyCount+ "/2"; } for ( const p of room.players ) { send( p.ws, { type:"room", code:room.code, count:room.players.length, readyText:text } ); } } /* ========================= START GAME ========================= */ function startGame(room) { if (room.started) return; if ( room.players.length !== 2 ) return; if ( !room.players.every( p => p.ready ) ) return; room.started=true; let number=3; broadcast( room, { type:"countdown", n:number } ); const timer = setInterval(()=>{ number--; if(number>0){ broadcast( room, { type:"countdown", n:number } ); } else{ clearInterval(timer); broadcast( room, { type:"start" } ); } },1000); } /* ========================= CONNECTION ========================= */ wss.on( "connection", ws => { const id = "p"+nextId++; const player = { id:id, ws:ws, name:"Noob", room:null, ready:false, x:0, y:0, z:0, r:0, hp:100, vy:0, coins:0, owned:[ "Noob" ], skin:"Noob" }; players.set( id, player ); send( ws, { type:"init", id:id, coins:0 } ); /* ========================= MESSAGES ========================= */ ws.on( "message", raw => { let m; try { m = JSON.parse( raw.toString() ); } catch { return; } /* CREATE / JOIN */ if ( m.type==="create" || m.type==="join" ) { if(player.room) return; let room; if( m.type==="create" ){ room={ code: createRoomCode(), players:[], started:false }; rooms.set( room.code, room ); } else{ room = rooms.get( m.code ); } if(!room){ send( ws, { type:"hit", text: "❌ Không tìm thấy phòng" } ); return; } if( room.players.length>=2 ){ send( ws, { type:"hit", text: "❌ Phòng đã đầy" } ); return; } player.name = m.name || "Noob"; player.room = room; player.ready=false; player.x = room.players.length===0 ? -5 : 5; player.z=0; player.hp=100; room.players.push( player ); updateRoom(room); } if(!player.room) return; const room = player.room; /* READY */ if( m.type==="ready" ){ if(room.started) return; player.ready = !!m.value; updateRoom(room); startGame(room); return; } /* MOVE */ if( m.type==="move" ){ if(!room.started) return; player.x += m.x * 0.42; player.z += m.z * 0.42; player.r = m.r || 0; player.x = Math.max( -16, Math.min( 16, player.x ) ); player.z = Math.max( -16, Math.min( 16, player.z ) ); } /* JUMP */ if( m.type==="jump" ){ if( room.started && player.y<=0.01 ){ player.vy=9; } } /* ATTACK */ if( m.type==="attack" ){ if(!room.started) return; for( const enemy of room.players ){ if(enemy===player) continue; const distance = Math.hypot( enemy.x-player.x, enemy.z-player.z ); if( distance<2.5 ){ enemy.hp -= 25; send( enemy.ws, { type:"hp", hp:enemy.hp } ); send( player.ws, { type:"hit", text:"⚔️ HIT!" } ); if( enemy.hp<=0 ){ player.coins += 25; enemy.hp=100; /* RESPAWN */ enemy.x = enemy.x<0 ? 5 : -5; enemy.z=0; player.x = player.x<0 ? -5 : 5; player.z=0; send( player.ws, { type:"coins", coins: player.coins } ); send( player.ws, { type:"hit", text: "🏆 KO! +25 COINS" } ); send( enemy.ws, { type:"hp", hp:100 } ); } } } } /* BUY SKIN */ if( m.type==="buy" ){ const cost = skinCost[ m.skin ]; if( cost===undefined ) return; if( player.owned.includes( m.skin ) ){ send( player.ws, { type:"hit", text: "✓ Bạn đã có skin này" } ); return; } if( player.coins < cost ){ send( player.ws, { type:"hit", text: "❌ Không đủ coins" } ); return; } player.coins -= cost; player.owned.push( m.skin ); player.skin = m.skin; send( player.ws, { type:"coins", coins: player.coins } ); send( player.ws, { type:"hit", text: "🛒 Mua "+ m.skin+ " thành công!" } ); } } ); /* ========================= DISCONNECT ========================= */ ws.on( "close", ()=>{ if(player.room){ const room = player.room; } ); } ); /* ========================= PHYSICS + STATE ========================= */ setInterval( ()=>{ for( const room of rooms.values() ){ for( const p of room.players ){ p.vy -= 0.45; p.y += p.vy * 0.05;
